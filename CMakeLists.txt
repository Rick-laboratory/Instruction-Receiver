cmake_minimum_required(VERSION 3.16)
project(InstructionReceiver LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This receiver requires Winsock and Win32 input APIs.")
endif()
add_library(receiver_core command.cpp UDP_Receiver.cpp)
target_include_directories(receiver_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(receiver_core PUBLIC cxx_std_17)
target_link_libraries(receiver_core PUBLIC ws2_32)
add_executable(instruction-receiver main.cpp Message_Interpreter.cpp)
target_link_libraries(instruction-receiver PRIVATE receiver_core user32)
include(CTest)
if(BUILD_TESTING)
    add_executable(receiver-test tests/receiver_test.cpp)
    target_link_libraries(receiver-test PRIVATE receiver_core)
    add_test(NAME protocol-and-loopback COMMAND receiver-test)
endif()
foreach(target receiver_core instruction-receiver receiver-test)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
